cmake_minimum_required(VERSION 3.20)
project(drm_odc_live LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(drm_odc_live src/main.cpp)
target_compile_options(drm_odc_live PRIVATE -Wall -Wextra -Wpedantic -Werror -O2)
find_package(Threads REQUIRED)
target_link_libraries(drm_odc_live PRIVATE Threads::Threads)
enable_testing()
add_test(NAME drm_odc_self_test COMMAND drm_odc_live --self-test)

add_executable(drm_runtime_descent src/runtime_descent.cpp)
target_compile_options(drm_runtime_descent PRIVATE -Wall -Wextra -Wpedantic -Werror -O2)
target_link_libraries(drm_runtime_descent PRIVATE Threads::Threads)
add_test(NAME drm_runtime_descent_integration COMMAND drm_runtime_descent --out ${CMAKE_BINARY_DIR}/runtime_test)
