cmake_minimum_required(VERSION 3.20)
project(drm_bytecode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(drm_bytecode src/bytecode.cpp)
target_compile_options(drm_bytecode PRIVATE -Wall -Wextra -Wpedantic -Werror -O3)
target_link_libraries(drm_bytecode PRIVATE Threads::Threads)
enable_testing()
add_test(NAME drm_bytecode_self_test COMMAND drm_bytecode --self-test)
